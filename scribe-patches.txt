function scribeNoteRowStatusClass(status) {
  if (status === 'uploaded') return 'scribe-note-row--graded'
  if (status === 'submitted') return 'scribe-note-row--submitted'
  return 'scribe-note-row--pending'
}

function gradeBreakdownTip(g) {
  const parts = []
  if (g.accuracy != null) parts.push(`Accuracy ${g.accuracy}`)
  if (g.completeness != null) parts.push(`Completeness ${g.completeness}`)
  if (g.terminology != null) parts.push(`Terminology ${g.terminology}`)
  if (g.formatting != null) parts.push(`Formatting ${g.formatting}`)
  return parts.length ? parts.join(' ┬╖ ') : 'Score breakdown from QPS review'
}
